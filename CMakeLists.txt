cmake_minimum_required(VERSION 3.25)

# CMP0141: CMAKE_MSVC_DEBUG_INFORMATION_FORMAT as a property instead of a
# hand-edited flag string. Must be set before project() initialises the flags.
cmake_policy(SET CMP0141 NEW)

list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake")
include(VcpkgSetup)

project(FrameCapture
    VERSION 0.1.0
    DESCRIPTION "Windows screen recorder"
    LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "FrameCapture is Windows-only (SPEC.md §2.2 item 2).")
endif()

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set_property(GLOBAL PROPERTY USE_FOLDERS ON)

if(NOT CMAKE_BUILD_TYPE AND NOT CMAKE_CONFIGURATION_TYPES)
    set(CMAKE_BUILD_TYPE Release CACHE STRING "" FORCE)
endif()

# All build outputs land in one place so run-dev and the packaging step do not
# have to know about generator-specific layouts.
foreach(_fc_cfg "" _DEBUG _RELEASE _RELWITHDEBINFO _MINSIZEREL)
    set(CMAKE_RUNTIME_OUTPUT_DIRECTORY${_fc_cfg} "${CMAKE_BINARY_DIR}/bin")
endforeach()
unset(_fc_cfg)

# ---------------------------------------------------------------------------
# Toolchain settings (SPEC.md §21.1)
# ---------------------------------------------------------------------------
if(WIN32)
    # NOMINMAX: windows.h defines min/max as macros, which breaks every
    # std::min/std::max call site with a C2589 that does not mention macros at all.
    # WIN32_LEAN_AND_MEAN: trims the include graph; add back any specific header a
    # translation unit needs rather than removing this.
    add_compile_definitions(NOMINMAX WIN32_LEAN_AND_MEAN)
endif()

if(MSVC)
    # Separate PDBs in every configuration. You cannot debug a user's crash dump
    # without them, so Release is not an exception.
    set(CMAKE_MSVC_DEBUG_INFORMATION_FORMAT "ProgramDatabase")
    add_link_options("/DEBUG:FULL")

    # /O2 /Gy /DNDEBUG come from CMake's Release defaults; /GL + /LTCG is IPO.
    add_compile_options("$<$<CONFIG:Release>:/Gy>")
    include(CheckIPOSupported)
    check_ipo_supported(RESULT FC_IPO_SUPPORTED OUTPUT FC_IPO_ERROR)
    if(FC_IPO_SUPPORTED)
        set(CMAKE_INTERPROCEDURAL_OPTIMIZATION_RELEASE ON)
    else()
        message(WARNING "LTCG unavailable, Release will not be link-time optimised: ${FC_IPO_ERROR}")
    endif()

    # /MP is a build-time-only win, but a large one on an 8-core part.
    add_compile_options("/MP")

    # /WX must not fail on warnings from FFmpeg, spdlog or toml++ headers. CMake
    # passes their include directories as -external:I; this makes that mean
    # something. Our own headers keep /W4.
    add_compile_options("/external:W0")
endif()

# ---------------------------------------------------------------------------
# Dependencies (vcpkg manifest mode -- see vcpkg.json)
# ---------------------------------------------------------------------------
find_package(FFMPEG REQUIRED)
find_package(spdlog CONFIG REQUIRED)
find_package(tomlplusplus CONFIG REQUIRED)
find_package(nlohmann_json CONFIG REQUIRED)
find_package(GTest CONFIG REQUIRED)

# vcpkg's FindFFMPEG is a variable-style module. Wrap it once, here, so that no
# other CMakeLists has to deal with raw ${FFMPEG_LIBRARIES}.
add_library(FrameCapture_FFmpeg INTERFACE)
add_library(FrameCapture::FFmpeg ALIAS FrameCapture_FFmpeg)
target_include_directories(FrameCapture_FFmpeg SYSTEM INTERFACE ${FFMPEG_INCLUDE_DIRS})
target_link_directories(FrameCapture_FFmpeg INTERFACE ${FFMPEG_LIBRARY_DIRS})
target_link_libraries(FrameCapture_FFmpeg INTERFACE ${FFMPEG_LIBRARIES})

# Check the headers rather than FFMPEG_<lib>_FOUND: the port's FindFFMPEG sets
# those with `set(var TRUE BOOL)` (a normal variable, not a cache entry) and then
# early-returns on any reconfigure once FFMPEG_FOUND is cached -- so they are
# absent the second time you configure. Header existence is true either way.
foreach(_fc_header libavcodec/avcodec.h libavformat/avformat.h libavutil/avutil.h libswresample/swresample.h)
    set(_fc_header_found FALSE)
    foreach(_fc_dir IN LISTS FFMPEG_INCLUDE_DIRS)
        if(EXISTS "${_fc_dir}/${_fc_header}")
            set(_fc_header_found TRUE)
            break()
        endif()
    endforeach()
    if(NOT _fc_header_found)
        message(FATAL_ERROR "FFmpeg header ${_fc_header} is missing from the vcpkg build.")
    endif()
endforeach()
unset(_fc_header)
unset(_fc_header_found)
unset(_fc_dir)

# Read the version out of FFmpeg's own generated header rather than trusting
# FFMPEG_VERSION. The port's module sets that as a normal variable inside a block
# it skips on every configure after the first, so it is empty on a reconfigure.
# ffversion.h is generated by the FFmpeg build itself and is authoritative.
set(FC_FFMPEG_VERSION "")
foreach(_fc_dir IN LISTS FFMPEG_INCLUDE_DIRS)
    if(EXISTS "${_fc_dir}/libavutil/ffversion.h")
        file(STRINGS "${_fc_dir}/libavutil/ffversion.h" _fc_version_line REGEX "define[ \t]+FFMPEG_VERSION")
        if(_fc_version_line MATCHES "\"([^\"]+)\"")
            set(FC_FFMPEG_VERSION "${CMAKE_MATCH_1}")
            break()
        endif()
    endif()
endforeach()
if(NOT FC_FFMPEG_VERSION AND FFMPEG_VERSION)
    set(FC_FFMPEG_VERSION "${FFMPEG_VERSION}")
endif()
unset(_fc_version_line)
unset(_fc_dir)

# A dependency that cannot report its version is a dependency we cannot
# identify in a crash report, so fail rather than print a blank.
foreach(_fc_pkg FC_FFMPEG spdlog tomlplusplus nlohmann_json GTest)
    if(NOT ${_fc_pkg}_VERSION)
        message(FATAL_ERROR "${_fc_pkg} was found but reported no version.")
    endif()
endforeach()
unset(_fc_pkg)

message(STATUS "FrameCapture: FFmpeg     ${FC_FFMPEG_VERSION}")
message(STATUS "FrameCapture: spdlog     ${spdlog_VERSION}")
message(STATUS "FrameCapture: toml++     ${tomlplusplus_VERSION}")
message(STATUS "FrameCapture: nlohmann   ${nlohmann_json_VERSION}")
message(STATUS "FrameCapture: GoogleTest ${GTest_VERSION}")

include(CompileShader)

add_subdirectory(engine/core)
add_subdirectory(engine/app)

include(CTest)
if(BUILD_TESTING)
    add_subdirectory(tests)
endif()
